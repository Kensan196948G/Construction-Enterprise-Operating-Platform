## 2. データベース設計

### t_employee（従業員マスタ - 共通マスタ）
```sql
CREATE TABLE t_employee (
    emp_id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    emp_code            VARCHAR(10) UNIQUE NOT NULL,
    name                VARCHAR(100) NOT NULL,
    name_kana           VARCHAR(100),
    department_id       UUID REFERENCES t_department(dept_id),
    position            VARCHAR(50),
    hire_date           DATE,
    entra_id            VARCHAR(200),  -- Entra ID UPN
    email               VARCHAR(200),
    phone               VARCHAR(20),
    status              VARCHAR(20) DEFAULT 'active',
    created_at          TIMESTAMPTZ DEFAULT NOW()
);
```

### t_qualification（資格管理）
```sql
CREATE TABLE t_qualification (
    qual_id             UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    emp_id              UUID REFERENCES t_employee(emp_id),
    qualification_name  VARCHAR(200) NOT NULL,
    qualification_code  VARCHAR(30),
    acquired_date       DATE,
    expiry_date         DATE,
    certificate_file    VARCHAR(500),
    ccus_linked         BOOLEAN DEFAULT FALSE,
    created_at          TIMESTAMPTZ DEFAULT NOW()
);
```

### t_invoice（インボイス管理）
```sql
CREATE TABLE t_invoice (
    invoice_id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    invoice_number      VARCHAR(30),
    issuer_name         VARCHAR(200),
    issuer_registration VARCHAR(20),  -- 適格請求書発行事業者登録番号
    amount              BIGINT,
    tax_amount          BIGINT,
    tax_rate            DECIMAL(4,2),
    invoice_date        DATE,
    payment_due_date    DATE,
    e_archive_path      VARCHAR(500),  -- 電帳法保存パス
    timestamp_token     VARCHAR(200),  -- タイムスタンプ
    status              VARCHAR(20),
    created_at          TIMESTAMPTZ DEFAULT NOW()
);
CREATE INDEX idx_invoice_search ON t_invoice(issuer_name, invoice_date, amount);
```

### t_workflow（稟議ワークフロー）
```sql
CREATE TABLE t_workflow (
    workflow_id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workflow_type       VARCHAR(30),  -- '稟議'/'支払承認'/'休暇申請'/'出張申請'
    title               VARCHAR(200),
    applicant_id        UUID REFERENCES t_employee(emp_id),
    content             JSONB,
    approval_route      JSONB,  -- [{approver_id, status, commented_at}]
    current_step        INT DEFAULT 1,
    status              VARCHAR(20) DEFAULT 'pending',
    desknet_sync_id     VARCHAR(50),  -- DeskNet's NEO連携ID
    created_at          TIMESTAMPTZ DEFAULT NOW()
);
```
